cmake_minimum_required(VERSION 3.20)
project(loadonce_redirectors LANGUAGES C)

if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "LoadOnce redirectors currently support Linux and Windows")
endif()

function(loadonce_redirector target)
    target_compile_features(${target} PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
        target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(WIN32)
        target_link_libraries(${target} PRIVATE shell32)
    endif()
endfunction()

add_executable(java redirector.c)
loadonce_redirector(java)
if(WIN32)
    add_executable(javaw WIN32 redirector.c)
    target_compile_definitions(javaw PRIVATE LOADONCE_GUI=1)
    loadonce_redirector(javaw)
    # Test-only Unicode boundary witness; distributions contain only java/javaw.
    add_executable(wide-argument-witness wide_argument_witness.c)
    loadonce_redirector(wide-argument-witness)
endif()
